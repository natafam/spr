package com.example.sprawdzian;

import android.os.Bundle;
import androidx.activity.EdgeToEdge;
import androidx.appcompat.app.AppCompatActivity;
import android.view.View;
import android.widget.ArrayAdapter;
import android.widget.Spinner;
import android.widget.TextView;

public class MainActivity extends AppCompatActivity {

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        EdgeToEdge.enable(this);
        setContentView(R.layout.activity_main);
        String[] options = getResources().getStringArray(R.array.options);
        Spinner dropdown = findViewById(R.id.list);
        ArrayAdapter<String> adapter = new ArrayAdapter<>(this, android.R.layout.simple_list_item_1, options);
        dropdown.setAdapter(adapter);
    }


    public void Sprawdz(View view) {
        TextView t1 = findViewById(R.id.text1);
        Spinner dropdown = findViewById(R.id.list);
        String selectedItem = (String) dropdown.getSelectedItem();
        if(selectedItem.equals("słodki")){
            t1.setText("Cukier jest słodki.");
        }
        else{
            t1.setText("Niestety wybrałeś błędną odpowiedź.");
        }


    }
}
